import { rect, render } from "svjeez";
import {
  Callout,
  adaptive,
  Focus,
  Marker,
  code,
  morph,
  readingTime,
  timeline,
  systemDark,
  systemLight,
  type CodeRun,
  type CodeTokens,
} from "svjeez/extras";

const { theme, css: colors } = adaptive({
  light: systemLight,
  dark: systemDark,
});

// A code explainer: two versions of a function, a timeline that names
// the beats, and annotations aimed at runs by id.
const kw = (text: string): CodeRun => ["keyword", text];
const fn = (text: string, id?: string): CodeRun => ["fn", text, id];
const p = (text: string, id?: string): CodeRun => ["punct", text, id];

const before: CodeTokens = [
  [
    kw("const"),
    " total ",
    p("="),
    " ",
    fn("sum", "call"),
    p("(prices);"),
  ],
  [kw("return"), " total", p(";")],
];
const after: CodeTokens = [
  [
    kw("const"),
    " total ",
    p("="),
    " ",
    fn("prices.reduce", "call"),
    p("((a, b) => a + b, 0);"),
  ],
  [kw("return"), " total", p(";")],
];

const mono = {
  x: 48,
  y: 64,
  fontSize: 18,
  lineHeight: 1.8,
  theme,
  gutter: true,
};
const one = code(before, mono);
const two = code(after, mono);

// readingTime sizes a beat to its caption. `change` is the beat where
// one version morphs into the other.
const tl = timeline(
  {
    intro: readingTime("A call to sum"),
    change: 3,
    outro: readingTime("Inline the loop"),
  },
  { gap: 0.4 },
);
const move = morph([one, two], { at: [tl.change.at], gutter: true });

export default function build(): string {
  return render(
    {
      width: 640,
      height: 200,
      duration: tl.end + 2,
      title: "Inline a helper",
      desc: "A call to sum is replaced by a reduce, with a marker on the call and a note after the change.",
      reducedMotion: { at: tl.end },
      css: `${colors}\n.sans text{font-family:system-ui,sans-serif}`,
    },
    <>
      <rect width="100%" height="100%" fill={theme.base00} />
      <Focus
        stops={[{ during: tl.intro, target: one.lines(1) }]}
        dim={0.35}
      >
        {move.node}
      </Focus>
      <Marker
        theme={theme}
        stops={[
          { during: tl.intro, target: one.get("call").pad(4) },
          { during: tl.change, target: two.get("call").pad(4) },
        ]}
      />
      <Callout
        theme={theme}
        target={two.lines(1)}
        side="bottom"
        during={tl.outro}
        columns={28}
        class="sans"
      >
        The loop now lives in the call site
      </Callout>
    </>,
  );
}
