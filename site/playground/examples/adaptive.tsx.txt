import { enter, ease, render } from "svjeez";
import { adaptive, systemDark, systemLight } from "svjeez/extras";

// adaptive() takes a light and a dark palette. `theme` holds CSS
// variables, and `css` defines them with a prefers-color-scheme
// media query, so the same file suits light and dark pages.
const { theme, css } = adaptive({
  light: systemLight,
  dark: systemDark,
});

export default function build(): string {
  return render(
    {
      width: 320,
      height: 140,
      duration: 4,
      title: "Adaptive",
      desc: "Three dots rise in turn, colored for light or dark mode.",
      css,
      reducedMotion: { at: 2 },
    },
    <>
      <rect width={320} height={140} fill={theme.base00} />
      {[theme.base08, theme.base09, theme.base0B].map((fill, i) => (
        <circle
          fill={fill}
          cx={110 + i * 50}
          cy={64}
          r={14}
          animation={enter(
            { at: 0.3 + i * 0.25, over: 0.5 },
            { y: 22, scale: 0.4 },
            ease.pop,
          )}
        />
      ))}
      <text
        fill={theme.base05}
        x={160}
        y={112}
        textAnchor="middle"
        fontFamily="system-ui, sans-serif"
        fontSize={13}
        animation={enter({ at: 1.2, over: 0.5 })}
      >
        Switch the preview between light and dark
      </text>
    </>,
  );
}
