import { draw, ease, enter, render } from "svjeez";
import { adaptive, systemDark, systemLight } from "svjeez/extras";

const { theme, css } = adaptive({
  light: systemLight,
  dark: systemDark,
});

// A curve draws itself on, then its end point pops in. draw() needs no
// path length.
export default function build(): string {
  return render(
    {
      width: 320,
      height: 140,
      duration: 4,
      title: "Draw-on curve",
      desc: "A line draws across the frame and ends in a dot.",
      css,
      reducedMotion: { at: 3 },
    },
    <>
      <rect width={320} height={140} fill={theme.base00} />
      <path d="M0 104H320M0 70H320M0 36H320" stroke={theme.base02} />
      <path
        d="M28 108C70 108 78 36 124 36S176 104 214 88S262 34 292 40"
        fill="none"
        stroke={theme.base0D}
        strokeWidth={4}
        strokeLinecap="round"
        animation={draw({ at: 0.4, over: 1.8 }, ease.inOut)}
      />
      <circle
        cx={292}
        cy={40}
        r={7}
        fill={theme.base0D}
        animation={enter(
          { at: 2.1, over: 0.4 },
          { scale: 0 },
          ease.pop,
        )}
      />
    </>,
  );
}
