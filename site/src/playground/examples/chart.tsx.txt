import { chart } from "@svjeez/charts";
import { render, adaptive, systemDark, systemLight } from "svjeez";
import { rect } from "svjeez/geometry";

const { theme, css } = adaptive({
  light: systemLight,
  dark: systemDark,
});

const months = ["Jan", "Feb", "Mar", "Apr", "May", "Jun"];
const stars = { Jan: 12, Feb: 19, Mar: 31, Apr: 44, May: 62, Jun: 87 };

// chart() fits the scales once; each series hands back the
// span it plays in, so the next one can start where it ends.
const c = chart({
  frame: rect(44, 20, 260, 130),
  x: { kind: "band", domain: months },
  data: Object.entries(stars),
  theme,
  font: "ui-monospace, monospace",
  fontSize: 10,
  grid: true,
});

const bars = c.bars({ at: 0.4, over: 1.3 }, stars, { stagger: 0.12 });
const trend = c.line({ at: bars.span.end, over: 1.2 }, Object.entries(stars), {
  curve: "monotone",
  dots: 3,
});

export default function build(): string {
  return render(
    {
      width: 330,
      height: 190,
      duration: 6,
      title: "Stars",
      desc: "Six bars grow month by month, then a line draws across their tops.",
      css,
      reducedMotion: { at: 5 },
    },
    <>
      <rect width={330} height={190} fill={theme.base00} />
      {c.node}
      {bars.node}
      {trend.node}
    </>,
  );
}
